' === RELATIONSHIPS ===

' OrderManager manages Orders
OrderManager "1" *-- "*" Order : manages

' OrderManager references Recipe pool
OrderManager "1" o-- "*" Recipe : references

' OrderManager spawns and tracks Customers
OrderManager "1" --> "*" CustomerGO : spawns
OrderManager "1" --> "*" Customer : tracks

' OrderManager owns CustomerQueueManager
OrderManager "1" *-- "1" CustomerQueueManager : owns

' OrderManager uses CounterSlot
OrderManager "1" o-- "1" CounterSlot : references

' Order relationships
OrderUIGO "1" *-- "1" Order : contains
Order "1" o-- "1" Recipe : references
Order "1" --> "1" OrderManager : notifies

' Customer relationships
CustomerGO "1" *-- "1" Customer : contains
Customer "1" o-- "1" Order : linked order
Customer "1" o-- "1" OrderManager : references
Customer "1" o-- "1" CounterSlot : checks
Customer "1" *-- "1" CustomerState : has state

' CustomerQueueManager relationships
CustomerQueueManager "1" *-- "*" Customer : manages queue
CustomerQueueManager "1" o-- "1" Customer : current at counter

' Recipe relationships
Recipe "1" o-- "*" CollectableType : ingredients
Recipe "1" o-- "1" CollectableType : foodType
Recipe "1" o-- "1" CookingFormula : formula

' CounterSlot relationships
CounterSlot "1" o-- "0..1" Collectable : currentItem

' Order events and interactions
Order --> Customer : OnOrderExpired\nevent
Customer --> Order : subscribes to event
Customer --> CounterSlot : checks for\nmatching item

' OrderManager external interactions
OrderManager --> PlayerStats : awards money
OrderManager --> DataManager : tracks statistics
OrderManager --> CookingTutorial : tutorial mode

' EndOfDayManager relationships
EndOfDayManager "1" ..> "1" EndOfDayManager : <<Singleton>>
EndOfDayManager "1" o-- "1" OrderManager : checks
EndOfDayManager "1" o-- "1" EndOfDayUI : references
EndOfDayManager "1" *-- "1" DayStats : owns
EndOfDayManager --> GameManager : gets current day
OrderManager --> EndOfDayManager : notifies when\nno valid recipes

' Fulfillment flow
CounterSlot --> OrderManager : item placed\ntriggers check
OrderManager --> Collectable : validates item type

@enduml
