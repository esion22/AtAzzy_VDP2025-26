' === RELATIONSHIPS ===

' Singleton pattern
AudioManager "1" ..> "1" AudioManager : <<Singleton>>

' AudioManager GameObject composition
AudioManagerGO "1" *-- "1" AudioManager : contains

' AudioManager owns multiple AudioSources
AudioManager "1" *-- "8" AudioSource : owns
AudioManager "1" o-- "many" AudioClip : references

' AudioManager subscribes to scene events
AudioManager --> SceneManager : subscribes to\nsceneLoaded
AudioManager --> Scene : uses

' Settings Menu relationships
SettingsMenuUI "1" *-- "1" SettingsMenu : contains
SettingsMenuUI "1" *-- "1" MusicSliderUI : contains
SettingsMenuUI "1" *-- "1" SFXSliderUI : contains
SettingsMenu "1" o-- "1" Slider : musicSlider
SettingsMenu "1" o-- "1" Slider : sfxSlider
SettingsMenu --> AudioManager : calls volume methods
